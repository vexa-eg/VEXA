/* ══════════════ STORAGE CALC ══════════════ */
function calcStorage(profile){
  if(!profile)return 0;
  let t=0;
  if(profile.bgBytes)t+=profile.bgBytes;
  (profile.links||[]).forEach(l=>{
    if(l.mediaBytes)t+=l.mediaBytes;
    if(l.coverBytes)t+=l.coverBytes;
    (l.items||[]).forEach(it=>{if(it.mediaBytes)t+=it.mediaBytes;});
  });
  return t;
}

function renderUpgrades(){
  const list=document.getElementById('upgrades-list');
  const all=Object.entries(_reqs);

  all.sort((a,b)=>{
    const ap=!a[1].status||a[1].status==='pending';
    const bp=!b[1].status||b[1].status==='pending';
    if(ap&&!bp)return -1;if(!ap&&bp)return 1;
    return (b[1].timestamp||0)-(a[1].timestamp||0);
  });
  list.innerHTML=all.map(([uid,req])=>{
    const st=req.status||'pending';
    const ts=req.timestamp?new Date(req.timestamp).toLocaleString('en-GB',{day:'numeric',month:'short',hour:'2-digit',minute:'2-digit'}):'–';
    const hasImg=req.receiptUrl&&req.receiptUrl.startsWith('http');
    return `<div class="req-card">
      <div class="req-top">
        <div>
          <div class="req-email">${escH(req.email||'No email provided')}</div>
          <div class="req-plan-lbl">Requesting: <span class="req-plan-val">${escH(req.plan||'Unknown plan')}</span></div>

      </div>
      ${req.note?`<div class="req-note">"${escH(req.note)}"</div>`:''}
      ${hasImg?`<img class="req-receipt" src="${escH(req.receiptUrl)}" onload="this.style.display='block'" onclick="openLbox('${escH(req.receiptUrl)}')" title="Click to zoom">
      <div class="receipt-hint">📎 Payment receipt — click to zoom</div>`:'<div class="receipt-hint" style="color:var(--sub);">No receipt uploaded.</div>'}
      ${st==='pending'
        ?`<div class="req-actions">
          <button class="btn-approve" onclick="handleReq('${uid}','approved')">✓ Approve & Activate</button>
          <button class="btn-reject" onclick="handleReq('${uid}','rejected')">✕ Reject</button>
        </div>`
        :`<div class="reviewed-note">Reviewed ${req.reviewedAt?new Date(req.reviewedAt).toLocaleDateString('en-GB',{day:'numeric',month:'short',year:'numeric'}):''}</div>`}
    </div>`;
  }).join('');

async function handleReq(uid,decision){
  const req=_reqs[uid];if(!req)return;
  try{
    if(decision==='approved'){
      const plan=req.plan||'pro';
      const maxBytes=(_PLAN_MB[plan]||100)*1024*1024;
      await Promise.all([
        _db.ref('upgrades/'+uid).update({status:'approved',reviewedAt:Date.now()}),
        _db.ref('config/'+uid).update({plan,maxBytes,updatedAt:Date.now(),updatedBy:_ADMIN_UID}),
        _db.ref('users/'+uid+'/_maxBytes').set(maxBytes)
      ]);
      _reqs[uid].status='approved';
      if(_users[uid]){_users[uid].config.plan=plan;_users[uid].config.maxBytes=maxBytes;}
      showToast('Approved & activated ✓','ok');
    }else{
      await _db.ref('upgrades/'+uid).update({status:'rejected',reviewedAt:Date.now()});
      _reqs[uid].status='rejected';
      showToast('Request rejected.','ok');
    }
    renderUpgrades();renderStats();renderTable();updateReqBadge();
  }catch(e){showToast('Error: '+e.message,'err');}
}

  const targets=Object.values(_users).filter(u=>(u.config.plan||'free')===plan);
  if(!targets.length){res.style.color='var(--orange)';res.textContent='No users on '+plan+' plan.';return;}
  res.style.color='var(--sub2)';res.textContent='Applying to '+targets.length+' users…';
  const maxBytes=mb*1024*1024;
  let done=0,failed=0;
  await Promise.allSettled(targets.map(async u=>{
    try{
      await Promise.all([
        _db.ref('config/'+u.uid+'/maxBytes').set(maxBytes),
        _db.ref('users/'+u.uid+'/_maxBytes').set(maxBytes)
      ]);
      _users[u.uid].config.maxBytes=maxBytes;done++;
    }catch(e){failed++;}
  }));
  renderTable();renderStats();
  res.style.color=failed?'var(--orange)':'var(--green)';
  res.textContent='Done: '+done+' updated'+(failed?' · '+failed+' failed':'')+'.';}

async function bulkSetFeature(enable){
  const plan=document.getElementById('bfeat-plan').value;
  const feat=document.getElementById('bfeat-feat').value;
  const res=document.getElementById('bfeat-result');
  const targets=Object.values(_users).filter(u=>(u.config.plan||'free')===plan);
  if(!targets.length){res.style.color='var(--orange)';res.textContent='No users on '+plan+' plan.';return;}
  res.style.color='var(--sub2)';res.textContent='Applying to '+targets.length+' users…';
  let done=0,failed=0;
  await Promise.allSettled(targets.map(async u=>{
    try{
      await _db.ref('config/'+u.uid+'/features/'+feat).set(enable);
      if(!_users[u.uid].config.features)_users[u.uid].config.features={};
      _users[u.uid].config.features[feat]=enable;done++;
    }catch(e){failed++;}
  }));
  res.style.color=failed?'var(--orange)':'var(--green)';
  res.textContent=(enable?'Enabled':'Disabled')+' for '+done+' users'+(failed?' · '+failed+' failed':'')+'.';}

function fmtBytes(b){
  const mb=b/(1024*1024);
  if(mb>=100)return Math.round(mb)+' MB';
  if(mb>=1)return mb.toFixed(1)+' MB';
  return Math.round(b/1024)+' KB';
}
